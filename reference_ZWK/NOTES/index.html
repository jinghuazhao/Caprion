split -l 10 -d -a 2 \ config/ZWK-reference-panel.unfinished.fp-manifest \ config/ZWK-reference-panel.batch- wc -l config/ZWK-reference-panel.batch-*

!/usr/bin/env bash

set -euo pipefail

next=3915954

for b in 00 01 02 03; do p="reports/A_retry\(b" python -u scripts/A_reconstruct_igs.py \ --manifest "config/ZWK-reference-panel.batch-\)b" \ --start-group-number "\(next" --out-prefix "\)p" \ 2>&1 | tee "${p}_console.log"

next=$(awk -F '\t' 'NR>1 && match($1,/[0-9]+$/) {
    n=substr($1,RSTART,RLENGTH)+0
    if(n>m)m=n
} END {if(m)print m+1; else exit 1}' \
    "${p}_reconstructed_igs.tsv")

echo "Batch $b done; next ID=$next"

done

inspect the featureXML summary

python - <<'PY' from pathlib import Path import numpy as np import pandas as pd import pyopenms as oms

path = Path("openms_pilot/szwk900404i19201xms1.featureXML") fm = oms.FeatureMap() oms.FeatureXMLFile().load(str(path), fm)

rows = [] for f in fm: rows.append({ "id": f.getUniqueId(), "mz": f.getMZ(), "rt_sec": f.getRT(), "intensity": f.getIntensity(), "charge": f.getCharge(), "quality": f.getOverallQuality(), "width": f.getWidth(), })

df = pd.DataFrame(rows) print("Features:", len(df)) print("\nCharge counts:") print(df["charge"].value_counts(dropna=False).sort_index().to_string()) print("\nNumeric summary:") print(df[["mz", "rt_sec", "intensity", "quality", "width"]] .describe(percentiles=[0.25, 0.5, 0.75]).to_string()) print("\nMissing/zero charge:", int((df["charge"] == 0).sum())) print("Missing/zero quality:", int((df["quality"] == 0).sum())) print("\nFirst 5 features:") print(df.head().to_string(index=False)) PY

save a concise QC summary

cd ~/Caprion/analysis/ZWK/reference_ZWK

python <<'PY' import pandas as pd from pathlib import Path

p = Path("openms_pilot/feature_overlap_qc_geometry.tsv") df = pd.read_csv(p, sep="\t")

fraction = df["shared_intensity_fraction"].fillna(0) shared = df["has_shared_peaks"].astype(bool)

summary = pd.DataFrame([ ("features_total", len(df)), ("features_with_shared_peaks", int(shared.sum())), ("fraction_features_with_shared_peaks", shared.mean()), ("mean_shared_intensity_fraction", fraction.mean()), ("median_shared_intensity_fraction", fraction.median()), ("p95_shared_intensity_fraction", fraction.quantile(0.95)), ("p99_shared_intensity_fraction", fraction.quantile(0.99)), ("maximum_shared_intensity_fraction", fraction.max()), ], columns=["metric", "value"])

out = Path("openms_pilot/feature_overlap_qc_summary.tsv") summary.to_csv(out, sep="\t", index=False)

print(summary.to_string(index=False)) print("\nSaved:", out) PY
